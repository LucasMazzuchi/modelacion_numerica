\section{Integración numérica}
\subsection{Método del trapecio}
\subsubsection{Método del Trapecio Simple}
Se utiliza para aproximar la integral en un solo intervalo $[a, b]$.
\begin{equation}
    I \approx \frac{h}{2} (f_0 + f_1)
\end{equation}
Donde el paso es $h = b - a$.
El error asociado es:
\begin{equation}
    E = -\frac{h^3}{12} f''(\xi) \quad \text{con } \xi \in (a, b)
\end{equation}

\subsubsection{Método del Trapecio Compuesto}
Se divide el intervalo total en $n$ subintervalos. Como los puntos intermedios pertenecen a dos trapecios adyacentes, sus valores se duplican.
\begin{equation}
    I \approx \frac{h}{2} \left[ f_0 + 2 \sum_{i=1}^{n-1} f_i + f_n \right]
\end{equation}
Donde el paso es $h = \frac{b - a}{n}$.
El error asociado es:
\begin{equation}
    E = -\frac{b - a}{12} h^2 f''(\xi) \quad \text{con } \xi \in (a, b)
\end{equation}

\subsection{Otras Fórmulas de Integración (Para tener a mano)}

\subsubsection{Método de Simpson 1/3 Simple}
Requiere 3 puntos (2 subintervalos).
\begin{equation}
    I \approx \frac{h}{3} (f_0 + 4f_1 + f_2)
\end{equation}
Donde $h = \frac{b - a}{2}$.
Error: $E = -\frac{h^5}{90} f^{(4)}(\xi)$

\subsubsection{Método de Simpson 1/3 Compuesto}
Requiere que el número de subintervalos $n$ sea par.
\begin{equation}
    I \approx \frac{h}{3} \left[ f_0 + 4 \sum_{\substack{i=1 \\ i \text{ impar}}}^{n-1} f_i + 2 \sum_{\substack{i=2 \\ i \text{ par}}}^{n-2} f_i + f_n \right]
\end{equation}
Donde $h = \frac{b - a}{n}$.
Error: $E = -\frac{b - a}{180} h^4 f^{(4)}(\xi)$

\subsubsection{Método de Simpson 3/8 Simple}
Requiere 4 puntos (3 subintervalos).
\begin{equation}
    I \approx \frac{3h}{8} (f_0 + 3f_1 + 3f_2 + f_3)
\end{equation}
Donde $h = \frac{b - a}{3}$.
Error: $E = -\frac{3h^5}{80} f^{(4)}(\xi)$

\subsubsection{Método de Simpson 3/8 Compuesto}
Requiere que el número de subintervalos $n$ sea múltiplo de 3.
\begin{equation}
    I \approx \frac{3h}{8} \left[ f_0 + 3 \sum_{\substack{i=1 \\ 3 \nmid i}}^{n-1} f_i + 2 \sum_{\substack{i=3 \\ 3 \mid i}}^{n-3} f_i + f_n \right]
\end{equation}
Donde $h = \frac{b - a}{n}$.
Error: $E = -\frac{b - a}{80} h^4 f^{(4)}(\xi)$
